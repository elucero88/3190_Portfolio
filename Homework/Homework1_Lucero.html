gghist <- function(x, bins=15){
  if(!is.numeric(x)){
    stop("Input contains non-numerical value(s)")
  }
  df <- data.frame(x=x)

ggbox <- function(x, y=NULL, horizontal=FALSE){

  #Case 1: Single Numeric Vector
  if(is.null(y)){
    if(!is.numeric(x)){
      stop("Inputs contain non numerical variables")
    }
    df <- data.frame(x=x)
#Horizontal or vertical plot?
    if(horizontal) {
      ggplot(df, aes(x=x))+
        geom_boxplot()+
        labs(x="Value")+
        theme_minimal()
    } else {
      ggplot(df, aes(y=x)) +
        geom_boxplot()+
        labs(y="Value")+
        theme_minimal()
    }
#Which variable is numeric? Which is categorical?
  } else {
    if(is.numeric(x) && !is.numeric(y)) {
      num <- x
      ctg <- y
    } else if (!is.numeric(x) && is.numeric(y)) {
      num <- y
      ctg <- x
    } else {
      stop("One input must be numeric and the other categorical")
    }

    df <- data.frame(num=num, ctg=as.factor(ctg))

#Makes boxplots horizontal for two box plots
    if(horizontal) {
      ggplot(df, aes(x=num, y=ctg))+
        geom_boxplot()+
        labs(x="Value", y="Group")+
        theme_minimal()
    }else{
      ggplot(df, aes(x=ctg, y=num))+
        geom_boxplot()+
        labs(x="Group", y="Value")+
        theme_minimal()
    }
  }
}
